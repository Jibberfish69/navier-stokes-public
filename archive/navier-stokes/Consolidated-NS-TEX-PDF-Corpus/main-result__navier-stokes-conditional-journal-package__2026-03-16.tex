\section{Main Result}

\begin{definition}[Classical target class]
The classical target class consists of classical solution trajectories of the original 3D incompressible Navier--Stokes equation on \([0,T)\times\mathbb{R}^3\), generated by smooth compactly supported divergence-free initial data, and for which each theorem-critical control mechanism is derived on the same equation surface with no essential dependence on modified-equation terms.
\end{definition}

\begin{theorem}[Conditional classical closure]
Let \(u\) lie in the classical target class. Assume:
\begin{enumerate}[leftmargin=*]
\item the coarse control layer is bounded in the form
  \[
  \sup_{0\le t<T}\|u(t)\|_{L^2(\mathbb{R}^3)}^2
  + 2\nu\int_0^T \|\nabla u(t)\|_{L^2(\mathbb{R}^3)}^2\,dt < \infty;
  \]
\item there exist dyadic projectors \(P_{\ge N}\), a high-frequency flux quantity \(\Pi_{\ge N}(t)\), and constants \(N_\ast\in\mathbb{N}\), \(C>0\), \(\delta>0\) such that
  \[
  |\Pi_{\ge N}(t)| \le C\,2^{-\delta N}
  \qquad\text{for all } N\ge N_\ast,\; 0\le t<T;
  \]
\item there is a classical approximation scheme \(u^{(n)}\to u\) strongly in \(L^2([0,T];L^2_{\mathrm{loc}}(\mathbb{R}^3))\) with
  \[
  u^{(n)}\otimes u^{(n)} \to u\otimes u
  \quad\text{in distributions on } (0,T)\times\mathbb{R}^3;
  \]
\item gradient control is forced by the same classical dynamics in the form
  \[
  \sup_{0\le t<T}\|\nabla u(t)\|_{L^2(\mathbb{R}^3)} < \infty;
  \]
\item no modified-equation correction term performs essential hidden work.
\end{enumerate}
Then the singularity pathway represented in this framework does not occur on \([0,T)\). In particular, any failure of classical continuation before time \(T\) must violate at least one of assumptions (2)--(5).
\end{theorem}

\begin{remark}[Claim boundary]
The theorem is conditional. This manuscript does not claim an unconditional proof of Navier-Stokes global regularity and does not claim the Clay Millennium problem is solved.
\end{remark}
