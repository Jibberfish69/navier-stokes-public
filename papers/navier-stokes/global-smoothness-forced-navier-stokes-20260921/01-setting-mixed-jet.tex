\section{The pressure-complete setting and mixed jet}
\label{sec:formal-setting-mixed-jet}

\begin{proposition}[Pressure normalization]
\label{prop:formal-pressure-normalization}
Let \(u,p,f\) be smooth on \(I\times\R^3\), let \(\nabla\cdot u=0\), and suppose
\[
 u(t)\in H^1_\sigma(\R^3),
 \qquad
 p(t)\in H^0(\R^3),
 \qquad f(t)\in L^1(\R^3)\cap H^1(\R^3)
 \qquad(t\in I),
\]
and
\begin{equation}
 u_t+(u\cdot\nabla)u+\nabla p=\nu\Delta u+f.
 \label{eq:formal-ns-system}
\end{equation}
Then
\begin{align}
 -\Delta p
 &=\partial_i\partial_j(u_i u_j)-\nabla\cdot f,
 \label{eq:formal-pressure-poisson}\\
 p
 &=R_iR_j(u_i u_j)-(-\Delta)^{-1}\nabla\cdot f,
 \label{eq:formal-pressure-riesz}\\
 (u\cdot\nabla)u+\nabla p
 &=\Pp\nabla\cdot(u\otimes u)+(I-\Pp)f.
 \label{eq:formal-pressure-projected}
\end{align}
The condition \(p(t)\in H^0(\R^3)\) fixes the pressure uniquely.
\end{proposition}

\begin{proof}
Incompressibility gives
\[
 (u\cdot\nabla)u_i=\partial_j(u_j u_i).
\]
Taking the divergence of \eqref{eq:formal-ns-system} cancels
\(\nabla\cdot u_t\) and \(\nabla\cdot\Delta u\), which yields
\eqref{eq:formal-pressure-poisson}.  Lemma~\ref{lem:riesz-leray}, applied
with \(m=0\), gives the force potential in \(H^0\) and
\[
 q:=R_iR_j(u_i u_j)-(-\Delta)^{-1}\nabla\cdot f\in H^0.
\]
It satisfies this Poisson identity and
\[
 (u\cdot\nabla)u+\nabla q
 =\Pp\nabla\cdot(u\otimes u)+(I-\Pp)f.
\]
If \(p_1,p_2\in H^0\) satisfy \eqref{eq:formal-pressure-poisson}, then
\(h=p_1-p_2\) satisfies
\(|\xi|^2\widehat h=0\).  Since \(\widehat h\in L^2\) and is supported at
\(\{0\}\), it vanishes.  Hence \(p=q\), which proves
\eqref{eq:formal-pressure-riesz} and \eqref{eq:formal-pressure-projected}.
\end{proof}

\begin{definition}[Maximal pressure-complete classical history]
\label{def:formal-maximal-history}
Fix \(\nu>0\), \(u_0\in\HsSigma(\R^3)\), and
\(f\in C^\infty([0,\infty);\mathcal S(\R^3;\R^3))\).  A
pressure-complete classical history generated by \((u_0,\nu,f)\) is a triple
\[
 (T_*,u,p),
 \qquad
 0<T_*\le\infty,
\]
such that, for every \(0<T<T_*\) and \(n,m\in\Nzero\),
\begin{align*}
 \partial_t^nu&\in C([0,T];H^m_\sigma(\R^3)),\\
 \partial_t^np&\in C([0,T];H^m(\R^3)),
\end{align*}
and
\begin{align}
 u_t+(u\cdot\nabla)u+\nabla p&=\nu\Delta u+f,
 \label{eq:formal-history-momentum}\\
 \nabla\cdot u&=0,
 \label{eq:formal-history-divergence}\\
 u(0)&=u_0,
 \label{eq:formal-history-datum}\\
 p(t)&=R_iR_j(u_i(t)u_j(t))
 -(-\Delta)^{-1}\nabla\cdot f(t).
 \label{eq:formal-history-pressure}
\end{align}
The history is maximal if no pressure-complete classical history
\((\widetilde T,\widetilde u,\widetilde p)\) generated by the same
\((u_0,\nu,f)\)
satisfies
\[
 \widetilde T>T_*,
 \qquad
 (\widetilde u,\widetilde p)|_{[0,T_*)}=(u,p).
\]
\end{definition}
\end{samepage}

\begin{definition}[Temporal and mixed velocity--pressure jets]
\label{def:formal-mixed-jet}
Let \((T_*,u,p)\) be a maximal pressure-complete classical history.  For
\(n\in\Nzero\) and \(\alpha\in\Nzero^3\), define
\[
 V_n:=\partial_t^nu,
 \qquad
 Q_n:=\partial_t^np,
 \qquad
 F_n:=\partial_t^nf,
 \qquad
 J_{n,\alpha}:=\partial_t^n\partial_x^\alpha u,
 \qquad
 \Pi_{n,\alpha}:=\partial_t^n\partial_x^\alpha p.
\]
Set also \(F_{n,\alpha}:=\partial_t^n\partial_x^\alpha f\).
For \(\beta\le\alpha\), define
\[
 \binom{\alpha}{\beta}
 :=\prod_{\ell=1}^3\binom{\alpha_\ell}{\beta_\ell},
 \qquad
 |\alpha|:=\alpha_1+\alpha_2+\alpha_3.
\]
\end{definition}

\begin{proposition}[Pressure-complete binomial recurrences]
\label{prop:formal-mixed-recurrences}
For every \(n\in\Nzero\) and \(\alpha\in\Nzero^3\),
\begin{align}
 Q_n
 &=R_iR_j
 \sum_{a=0}^{n}\binom na V_{a,i}V_{n-a,j}
 -(-\Delta)^{-1}\nabla\cdot F_n,
 \label{eq:formal-temporal-pressure-row}\\
 V_{n+1}
 &=
 \nu\Delta V_n
 -\sum_{a=0}^{n}\binom na(V_a\cdot\nabla)V_{n-a}
 -\nabla Q_n+F_n,
 \label{eq:formal-temporal-velocity-row}\\
 \Pi_{n,\alpha}
 &=R_iR_j
 \sum_{a=0}^{n}\sum_{\beta\le\alpha}
 \binom na\binom{\alpha}{\beta}
 J_{a,\beta,i}J_{n-a,\alpha-\beta,j}
 -\partial_x^\alpha(-\Delta)^{-1}\nabla\cdot F_n,
 \label{eq:formal-mixed-pressure-row}\\
 J_{n+1,\alpha}
 &+
 \sum_{a=0}^{n}\sum_{\beta\le\alpha}
 \binom na\binom{\alpha}{\beta}
 (J_{a,\beta}\cdot\nabla)
 J_{n-a,\alpha-\beta}
 +\nabla\Pi_{n,\alpha}
 =\nu\Delta J_{n,\alpha}+F_{n,\alpha},
 \label{eq:formal-mixed-velocity-row}\\
 \nabla\cdot J_{n,\alpha}&=0.
 \label{eq:formal-mixed-divergence-row}
\end{align}
\end{proposition}

\begin{proof}
Apply \(\partial_t^n\) to \eqref{eq:formal-history-pressure} and use
Lemma~\ref{lem:mixed-leibniz} with \(\alpha=0\) to obtain
\eqref{eq:formal-temporal-pressure-row}.  Apply \(\partial_t^n\) to
\eqref{eq:formal-history-momentum} to obtain
\eqref{eq:formal-temporal-velocity-row}.  Apply
\(\partial_x^\alpha\) to both identities, use
Lemma~\ref{lem:mixed-leibniz}, and commute the derivatives with \(R_iR_j\)
to obtain \eqref{eq:formal-mixed-pressure-row} and
\eqref{eq:formal-mixed-velocity-row}.  Differentiation of
\eqref{eq:formal-history-divergence} gives
\eqref{eq:formal-mixed-divergence-row}.
\end{proof}

\paragraph{The forced datum and its derivative rows.}
The initial velocity, viscosity, and complete prescribed force specify all
inputs to these recurrences.  Pressure is determined by the velocity products
and the gradient part of that same force.  Every temporal and spatial force
derivative remains in its corresponding row.  In the kinetic-energy
calculation of Proposition~\ref{prop:kinetic-energy}, transport and pressure
have zero global pairing with the velocity while the external work remains.
At higher Sobolev orders, the full transport--pressure contribution and force
work are retained in Proposition~\ref{prop:completed-height-recurrence}.
These are the governing rows and energy identities used below.

\begin{definition}[Canonical pressure constant]
\label{def:formal-canonical-bilinear-constants}
For vector fields \(v,w\), set
\[
 \mathcal Q(v,w):=\sum_{i,j=1}^3R_iR_j(v_i w_j).
\]
For \(m\in\Nzero\), define
\[
 C_m^{\rm p}:=
 \sup_{\substack{
 v,w\in H^{m+1}(\R^3;\R^3)\setminus\{0\}}}
 \frac{
 \norm{\mathcal Q(v,w)}_{H^m}
 +\norm{\nabla\mathcal Q(v,w)}_{H^{m-1}}}
 {\norm v_{H^{m+1}}\norm w_{H^{m+1}}}.
\]
\end{definition}

\begin{lemma}
\label{lem:formal-canonical-bilinear-constants}
For every \(m\in\Nzero\),
\[
 C_m^{\rm p}\le2C_m^{\rm alg}.
\]
\end{lemma}
\begin{proof}
Regard \((v_i w_j)_{i,j}\) as one tensor.  The contraction
\(F\mapsto R_iR_jF_{ij}\) has Fourier-multiplier norm one from the tensor
\(H^m\) norm to scalar \(H^m\).  Lemma~\ref{lem:sobolev-product} and
\(\norm{\nabla h}_{H^{m-1}}\le\norm h_{H^m}\) give
\[
 \norm{\mathcal Q(v,w)}_{H^m}
 +\norm{\nabla\mathcal Q(v,w)}_{H^{m-1}}
 \le2C_m^{\rm alg}\norm v_{H^{m+1}}\norm w_{H^{m+1}}.
\]
\end{proof}

\begin{lemma}[Initial pressure-complete jet]
\label{lem:formal-initial-jet}
Set
\[
 V_n^0:=V_n(0),
 \qquad
 Q_n^0:=Q_n(0),
 \qquad
 J_{n,\alpha}^0:=J_{n,\alpha}(0),
 \qquad
 \Pi_{n,\alpha}^0:=\Pi_{n,\alpha}(0).
\]
Write \(F_n^0:=F_n(0)\).
Then
\begin{align}
 V_0^0&=u_0,
 \label{eq:formal-initial-base}\\
 Q_n^0
 &=R_iR_j
 \sum_{a=0}^{n}\binom naV_{a,i}^0V_{n-a,j}^0
 -(-\Delta)^{-1}\nabla\cdot F_n^0,
 \label{eq:formal-initial-pressure}\\
 V_{n+1}^0
 &=
 \nu\Delta V_n^0
 -\sum_{a=0}^{n}\binom na
 (V_a^0\cdot\nabla)V_{n-a}^0
 -\nabla Q_n^0+F_n^0.
 \label{eq:formal-initial-velocity}
\end{align}
For every \(n,m\in\Nzero\) and \(\alpha\in\Nzero^3\),
\[
 V_n^0,Q_n^0,J_{n,\alpha}^0,\Pi_{n,\alpha}^0\in H^m(\R^3).
\]
\par
\end{lemma}

\begin{proof}
Equations \eqref{eq:formal-initial-base}--\eqref{eq:formal-initial-velocity}
are \eqref{eq:formal-temporal-pressure-row} and
\eqref{eq:formal-temporal-velocity-row} at \(t=0\).  Assume
\(V_0^0,\ldots,V_n^0\in\bigcap_{m\ge0}H^m\).  Lemmas
\ref{lem:sobolev-product} and \ref{lem:riesz-leray} give
\[
 Q_n^0\in\bigcap_{m\ge0}H^m,
 \qquad
 V_{n+1}^0\in\bigcap_{m\ge0}H^m.
\]
Induction starts from \(u_0\in\HsSigma\), while every \(F_n^0\) is
Schwartz by the prescribed force class.  Spatial differentiation and
\eqref{eq:formal-mixed-pressure-row} give the claims for
\(J_{n,\alpha}^0\) and \(\Pi_{n,\alpha}^0\).
\end{proof}

\begin{theorem}[Finite-row pressure estimates]
\label{thm:formal-finite-row-pressure}
Let \(I=(0,T)\subset(0,T_*)\), let \(n,M\in\Nzero\), and suppose
\[
 V_a\in L^2(I;H^{M+1}(\R^3))
 \qquad(0\le a\le n).
\]
For \(\alpha\in\Nzero^3\), define the prescribed pressure-source budget
\begin{equation}
 \mathfrak F^{\rm p}_{n,\alpha,M}(I)
 :=\norm{\partial_x^\alpha(-\Delta)^{-1}\nabla\cdot F_n}_{L^1(I;H^M)}
 +\norm{\nabla\partial_x^\alpha(-\Delta)^{-1}\nabla\cdot F_n}_{L^1(I;H^{M-1})}.
 \label{eq:force-pressure-budget}
\end{equation}
It is finite by \eqref{eq:force-pressure-potential-bound} and the prescribed
force class.
Then
\begin{align}
 &\norm{Q_n}_{L^1(I;H^M)}
 +\norm{\nabla Q_n}_{L^1(I;H^{M-1})}
 \notag\\
 &\qquad\le
 C_M^{\rm p}\sum_{a=0}^{n}\binom na
 \norm{V_a}_{L^2(I;H^{M+1})}
 \norm{V_{n-a}}_{L^2(I;H^{M+1})}
 +\mathfrak F^{\rm p}_{n,0,M}(I).
 \label{eq:formal-finite-pressure-L1}
\end{align}
\begin{samepage}
If
\[
 \mathcal V_{n,M}(I)^2
 :=\sum_{a=0}^{n}
 \norm{V_a}_{L^2(I;H^{M+1})}^2,
\]
then
\begin{equation}
 \norm{Q_n}_{L^1(I;H^M)}
 +\norm{\nabla Q_n}_{L^1(I;H^{M-1})}
 \le C_M^{\rm p}2^n\mathcal V_{n,M}(I)^2
 +\mathfrak F^{\rm p}_{n,0,M}(I).
 \label{eq:formal-finite-pressure-compressed}
\end{equation}
\end{samepage}
For \(\alpha\in\Nzero^3\), assume
\[
 J_{a,\beta}\in L^2(I;H^{M+1}(\R^3))
 \qquad
 (0\le a\le n,\ \beta\le\alpha).
\]
Then
\begin{align}
 &\norm{\Pi_{n,\alpha}}_{L^1(I;H^M)}
 +\norm{\nabla\Pi_{n,\alpha}}_{L^1(I;H^{M-1})}
 \notag\\
 &\quad\le
 C_M^{\rm p}
 \sum_{a=0}^{n}\sum_{\beta\le\alpha}
 \binom na\binom{\alpha}{\beta}
 \norm{J_{a,\beta}}_{L^2(I;H^{M+1})}
 \norm{J_{n-a,\alpha-\beta}}_{L^2(I;H^{M+1})}
 +\mathfrak F^{\rm p}_{n,\alpha,M}(I).
 \label{eq:formal-finite-mixed-pressure-L1}
\end{align}
If
\[
 \mathcal J_{n,\alpha,M}(I)^2
 :=
 \sum_{a=0}^{n}\sum_{\beta\le\alpha}
 \norm{J_{a,\beta}}_{L^2(I;H^{M+1})}^2,
\]
then
\begin{equation}
 \norm{\Pi_{n,\alpha}}_{L^1(I;H^M)}
 +\norm{\nabla\Pi_{n,\alpha}}_{L^1(I;H^{M-1})}
 \le
 C_M^{\rm p}2^{n+|\alpha|}
 \mathcal J_{n,\alpha,M}(I)^2
 +\mathfrak F^{\rm p}_{n,\alpha,M}(I).
 \label{eq:formal-finite-mixed-pressure-compressed}
\end{equation}
\end{theorem}

\begin{proof}
Lemmas~\ref{lem:sobolev-product} and \ref{lem:riesz-leray} give
\begin{equation}
 \norm{R_iR_j(v_i w_j)}_{H^M}
 +\norm{\nabla R_iR_j(v_i w_j)}_{H^{M-1}}
 \le C_M^{\rm p}\norm v_{H^{M+1}}\norm w_{H^{M+1}}.
 \label{eq:formal-pressure-bilinear}
\end{equation}
Equations \eqref{eq:formal-temporal-pressure-row} and
\eqref{eq:formal-pressure-bilinear} give, for almost every \(t\in I\),
\begin{align*}
 \norm{Q_n(t)}_{H^M}
 +\norm{\nabla Q_n(t)}_{H^{M-1}}
 &\le
 C_M^{\rm p}\sum_{a=0}^{n}\binom na
 \norm{V_a(t)}_{H^{M+1}}
 \norm{V_{n-a}(t)}_{H^{M+1}}\\
 &\quad+\norm{(-\Delta)^{-1}\nabla\cdot F_n(t)}_{H^M}\\
 &\quad+\norm{\nabla(-\Delta)^{-1}\nabla\cdot F_n(t)}_{H^{M-1}}.
\end{align*}
Integration and Cauchy--Schwarz prove
\eqref{eq:formal-finite-pressure-L1}, with the two force-potential terms
collected exactly in \(\mathfrak F^{\rm p}_{n,0,M}(I)\).  For nonnegative \(x_a\),
\[
 \sum_{a=0}^{n}\binom nax_ax_{n-a}
 \le
 \sum_{a=0}^{n}\binom nax_a^2
 \le
 2^n\sum_{a=0}^{n}x_a^2.
\]
This proves \eqref{eq:formal-finite-pressure-compressed}.

Apply \eqref{eq:formal-pressure-bilinear} to each summand in
\eqref{eq:formal-mixed-pressure-row}.  Integration and Cauchy--Schwarz give
\eqref{eq:formal-finite-mixed-pressure-L1}; the differentiated prescribed
pressure source is \(\mathfrak F^{\rm p}_{n,\alpha,M}(I)\).  The involution
\[
 (a,\beta)\longmapsto(n-a,\alpha-\beta)
\]
preserves the weights
\(\binom na\binom{\alpha}{\beta}\), and
\[
 \sum_{a=0}^{n}\sum_{\beta\le\alpha}
 \binom na\binom{\alpha}{\beta}
 =2^{n+|\alpha|}.
\]
In particular, every individual weight is at most \(2^{n+|\alpha|}\).
The inequality \(2xy\le x^2+y^2\) proves
\eqref{eq:formal-finite-mixed-pressure-compressed}.
\end{proof}


\subsection{The datum-generated whole-terminal intersection}
\label{sec:independent-intersection}

Start with \(u_0\in\HsSigma(\R^3)\), \(\nu>0\), and the prescribed
\(f\in C^\infty([0,\infty);\mathcal S(\R^3;\R^3))\).
Theorem~\ref{thm:local-restart} constructs a pressure-normalized classical
history on a positive interval. Uniqueness identifies the local solutions
on their overlaps; their compatible union gives the maximal history of
Definition~\ref{def:formal-maximal-history}.

The datum fixes the initial velocity row. The pressure formula in
Proposition~\ref{prop:formal-mixed-recurrences} fixes its normalized pressure,
and the velocity recurrence fixes the next temporal row. Repetition and
spatial differentiation produce the complete initial mixed jet. Along the
history these are the actual derivatives of \(u,p,f\), with the complete
transport sum, pressure, viscosity, and force row retained. These finite jet
realizations construct a compatible intersection on every strict horizon.
The retained datum-generated construction carries every finite selection to
the same proposed terminal time. The critical-height and rectangle identities
begin independently from the same mixed jet in the next section.

\begin{theorem}[Datum-generated strict-horizon mixed-jet intersection]
\label{thm:strict-datum-intersection}
Let \(\nu>0\), \(u_0\in\HsSigma(\R^3)\), and
\(f\in C^\infty([0,\infty);\mathcal S(\R^3;\R^3))\). Let
\((T_*,u,p)\) be the maximal pressure-complete classical history generated
by \((u_0,\nu,f)\). For every \(0<T<T_*\),
\begin{equation}
 u\in\bigcap_{r,m\in\Nzero}H^r((0,T);H^m_\sigma(\R^3)).
 \label{eq:strict-independent-mixed-intersection}
\end{equation}
All velocity, pressure, and force coordinates are the actual mixed
derivatives of this one history, obey
Proposition~\ref{prop:formal-mixed-recurrences}, and agree under restriction.
\end{theorem}

\begin{proof}
Write \(A_n=V_n(0)\) and \(P_n=Q_n(0)\). The complete initial recurrence is
\[
 \begin{aligned}
 A_0&=u_0,\\
 P_n&=R_iR_j\sum_{a=0}^n\binom na(A_a)_i(A_{n-a})_j
      -(-\Delta)^{-1}\nabla\cdot F_n(0),\\
 A_{n+1}&=\nu\Delta A_n
 -\sum_{a=0}^n\binom na(A_a\cdot\nabla)A_{n-a}
 -\nabla P_n+F_n(0).
 \end{aligned}
\]
Lemma~\ref{lem:formal-initial-jet} shows inductively that every selected
velocity and pressure row lies in every finite Sobolev space.

Fix \(T<T_*\) and finite \(r,m\). Apply the forced local construction
simultaneously to the finite family
\[
 \bigl\{\partial_t^n\partial_x^\alpha u:
 0\le n\le r,\ |\alpha|\le m\bigr\},
\]
using the displayed recurrence at the initial face. Higher-order persistence
realizes these fields as the actual derivatives of one velocity. Whenever
two finite constructions overlap in indices or time, local uniqueness
identifies their common coordinates. On the compact interval \([0,T]\),
persistence gives
\[
 \sum_{n=0}^r\|V_n\|_{L^2(0,T;H^m)}^2<\infty.
\]
Taking every finite \((r,m)\) proves
\eqref{eq:strict-independent-mixed-intersection}. The pressure formula and
Theorem~\ref{thm:formal-finite-row-pressure} place each selected pressure row
in its stated strict-horizon Sobolev space. Restriction compatibility makes
the resulting family one mixed jet of one velocity.
\end{proof}

\begin{theorem}[Datum-generated whole-terminal forced intersection]
\label{thm:datum-intersection}
Let \(\nu>0\), \(u_0\in\HsSigma(\R^3)\), and
\(f\in C^\infty([0,\infty);\mathcal S(\R^3;\R^3))\). Let
\((T_*,u,p)\) be the maximal pressure-complete classical history generated
by the single forced datum \((u_0,\nu,f)\). If \(T_*<\infty\), then, with
\(I=(0,T_*)\),
\begin{equation}
 u\in\bigcap_{r,m\in\Nzero}H^r(I;H^m_\sigma(\R^3)).
 \label{eq:independent-mixed-intersection}
\end{equation}
Equivalently, every fixed finite \((r,m)\) satisfies
\begin{equation}
 C_{r,m}[u_0,\nu,f;I]
 :=\sum_{n=0}^r\|V_n\|_{L^2(I;H^m)}^2<\infty.
 \label{eq:direct-datum-terminal-membership}
\end{equation}
More precisely, the datum-generated adjacent-row family satisfies, for every
finite \((N,M)\),
\begin{equation}
 \mathfrak P_{N,M}[u_0,\nu,f;I]
 :=\sum_{n=0}^{N}\left(
 \|V_n\|_{L^2(I;H^{M+1})}^2
 +\|V_{n+1}\|_{L^2(I;H^{M-1})}^2
 \right)<\infty,
 \label{eq:datum-terminal-adjacent-producer}
\end{equation}
where negative spatial indices are understood in the Bessel-potential sense.
The complete compatible record is
\begin{equation}
 \cI[u_0,\nu,f]:=
 \bigl(J_{n,\alpha},\Pi_{n,\alpha},F_{n,\alpha}\bigr)_{n\in\Nzero,\,\alpha\in\Nzero^3},
 \quad
 J_{n,\alpha}=\partial_x^\alpha\partial_t^nu,\quad
 \Pi_{n,\alpha}=\partial_x^\alpha\partial_t^np,\quad
 F_{n,\alpha}=\partial_x^\alpha\partial_t^nf,
 \label{eq:independent-intersection-record}
\end{equation}
with
\[
 J_{n,\alpha}\in L^2(I;H^m),\qquad
 \Pi_{n,\alpha}\in L^1(I;H^m),\qquad
 F_{n,\alpha}\in C([0,T_*];H^m)
\]
for every finite \((n,m,\alpha)\). Every coordinate is generated by the same
initial forced jet, agrees with every lower finite projection, and obeys the
complete recurrences of Proposition~\ref{prop:formal-mixed-recurrences}.
\end{theorem}

\begin{proof}[Proof by forced finite-row transfer]
The prescribed data first generate the complete initial jet:
\[
 \begin{aligned}
 A_0&=u_0,\\
 P_n&=R_iR_j\sum_{a=0}^n\binom na(A_a)_i(A_{n-a})_j
      -(-\Delta)^{-1}\nabla\cdot F_n(0),\\
 A_{n+1}&=\nu\Delta A_n
 -\sum_{a=0}^n\binom na(A_a\cdot\nabla)A_{n-a}
 -\nabla P_n+F_n(0).
 \end{aligned}
\]
Lemma~\ref{lem:formal-initial-jet} places every finite row in every finite
spatial Sobolev space.  Fix \((N,M)\).  Apply the datum-generated
whole-terminal finite-row construction of \cite[Section~5]{Birnie2026Spine}
to the complete forced equation graph
\begin{equation}
 \mathscr E_{n,\alpha}(J,\Pi;F)
 :=J_{n+1,\alpha}
 +\sum_{a=0}^{n}\sum_{\beta\le\alpha}
  \binom na\binom{\alpha}{\beta}
  (J_{a,\beta}\cdot\nabla)J_{n-a,\alpha-\beta}
 +\nabla\Pi_{n,\alpha}
 -\nu\Delta J_{n,\alpha}-F_{n,\alpha}=0.
 \label{eq:forced-finite-equation-graph}
\end{equation}
The velocity, nonlinear, pressure, and viscous rows are the same finite
pressure-complete rows used in that construction.  No nonlinear term is
canceled or removed.  The forced graph adds only the prescribed coordinate
\(F_{n,\alpha}\) and its normalized pressure potential.  At every fixed order
their required contributions are finite.  In the Sobolev row,
\begin{equation}
 2\left|\langle F_n,V_n\rangle_{H^M}\right|
 \le \nu\norm{V_n}_{H^{M+1}}^2
     +\nu^{-1}\norm{F_n}_{H^{M-1}}^2,
 \label{eq:forced-finite-row-source-payment}
\end{equation}
and the gradient component has the finite pressure-source budget
\(\mathfrak F^{\rm p}_{n,\alpha,M}(I)\) from
\eqref{eq:force-pressure-budget}.  Smoothness in time with Schwartz spatial
values makes both source budgets finite on \([0,T_*]\).  Thus the forced
finite-row construction has exactly the same terminal output, enlarged by
finite prescribed-source coordinates:
\eqref{eq:datum-terminal-adjacent-producer}.

Enlarging \((N,M)\) preserves every lower coordinate.  The triangular temporal
recurrence, the spatial Leibniz rule, pressure normalization, and the prescribed
force jet all commute with restriction, while overlap uniqueness identifies
every finite record with the same maximal forced history.  Hence these records
form one compatible whole-terminal family rather than separately chosen
solutions.

Fix finite \((r,m)\), choose \(N\ge r\) and \(M\ge m\), and project that
family.  Sobolev nesting in \eqref{eq:datum-terminal-adjacent-producer} gives
\(V_n\in L^2(I;H^m)\) for \(0\le n\le r\), proving
\eqref{eq:direct-datum-terminal-membership} and
\eqref{eq:independent-mixed-intersection}.  Spatial differentiation supplies
every \(J_{n,\alpha}\) from a sufficiently high projection.  For the normalized
pressure rows,
\[
 \|Q_n\|_{L^1(I;H^m)}
 \le C_m^{\rm p}\sum_{a=0}^n\binom na
 \|V_a\|_{L^2(I;H^{m+1})}
 \|V_{n-a}\|_{L^2(I;H^{m+1})}
 +\mathfrak F^{\rm p}_{n,0,m}(I),
\]
and the right-hand side is finite by the same rectangle and force budget.
The force rows already belong to every stated finite space.  Compatibility
therefore yields the single record
\eqref{eq:independent-intersection-record} and completes the proof.
\end{proof}
