% Original V6 file-backed section wrapper.
% Usage: \VolSection{<section title>}{<relative-file.tex>}
\newcommand{\VolSection}[2]{%
  \ifToEGroupHold
    \ToEGroupHoldfalse
  \else
    \ifToEVolumeHold
      \ToEVolumeHoldfalse
    \else
      \ifVolSectionNewpage
        \newpage
      \fi
    \fi
  \fi
  \section{#1}%
  \subimport{./}{#2}%
}

% Custom TOC line for group rows: italic text, no page number.
\makeatletter
\newcommand{\l@vgroup}[2]{%
  \addvspace{0.7\baselineskip}%
  {\noindent\itshape\small #1}\par
}
\providecommand{\toclevel@vgroup}{0}
\makeatother

% Heading size controls.
\newcommand{\HeadingSizeVolume}{\fontsize{18pt}{22pt}\selectfont}
\newcommand{\HeadingSizeGroup}{\fontsize{16.5pt}{20.5pt}\selectfont}
\newcommand{\HeadingSizeSection}{\fontsize{15pt}{18.5pt}\selectfont}
\newcommand{\HeadingSizeSubsection}{\fontsize{13.5pt}{16.5pt}\selectfont}
\newcommand{\HeadingSizeSubsubsection}{\fontsize{13pt}{15.5pt}\selectfont}

% On-page group divider, also written into the TOC.
\newcommand{\volumegroup}[1]{%
  \ifToEVolumeHold
    \ToEVolumeHoldfalse
  \else
    \ifVolGroupNewpage
      \clearpage
    \fi
  \fi
  \ToEGroupHoldtrue
  \par\vspace{1.8\baselineskip}%
  \phantomsection
  \addcontentsline{toc}{vgroup}{#1}%
  {\centering\bfseries\itshape\HeadingSizeGroup #1\par}%
  \vspace{0.6\baselineskip}%
}

% Keep TOC pages plain.
\pretocmd{\tableofcontents}{\pagestyle{plain}\markboth{}{}}{}{}

% Top-level part macro: compact on-page heading plus V6-style TOC row.
% Usage: \Volume{<major part title>}
\makeatletter
\newcommand{\Volume}[2][]{%
  \clearpage
  \refstepcounter{part}%
  \phantomsection
  \if\relax\detokenize{#1}\relax\else\label{#1}\fi
  \addcontentsline{toc}{part}{Volume~\thepart\ \textemdash\ #2}%
  \par\vspace{1.0ex}%
  \noindent{\centering\bfseries\HeadingSizeVolume Volume \thepart\ -- \ #2\par}%
  \vspace{0.8ex}%
  \ToEVolumeHoldtrue
}
\makeatother

\titleformat{\subsubsection}[block]{\bfseries\HeadingSizeSubsubsection}{\thesubsubsection}{1em}{}
