  \par\vspace{1.8\baselineskip}%

  {\centering\bfseries\itshape\HeadingSizeGroup #1\par}%
  \vspace{0.6\baselineskip}%
}

  \noindent{\centering\bfseries\HeadingSizeVolume Volume \thepart\ -- \ #2\par}%
  \vspace{0.8ex}%
  \ToEVolumeHoldtrue
}

\section{Formal Environment Examples}\label{sec:FormalEnvironmentExamples}
This section demonstrates the formal environment naming convention inside the
custom volume structure. The formal cross-reference rule makes
\cref{def:SampleObject} and \cref{thm:SampleConclusion} read as noun phrases
from the canonical PascalCase label stem. Section pointers can still use the
section sign, as in \Sref{sec:FormalEnvironmentExamples}.

\begin{definition}[sample object]\label{def:SampleObject}
A sample object is a placeholder mathematical item used only to show the
environment heading, optional title, counter, spacing, and cross-reference form.
\end{definition}

\begin{notationenv}[sample notation]\label{not:SampleNotation}
For a finite list of sample objects, write \(X_1,\ldots,X_n\), and let
\(P(X_i)\) denote the placeholder property attached to \(X_i\).
\end{notationenv}

\begin{assumption}[sample regularity hypothesis]\label{as:SampleRegularityHypothesis}
Each \(X_i\) carries enough structure for the statement \(P(X_i)\) to be
evaluated inside the present section.
\end{assumption}

\begin{theorem}[sample conclusion]\label{thm:SampleConclusion}
Under \cref{as:SampleRegularityHypothesis}, the object introduced in
\cref{def:SampleObject} has a well-defined placeholder property.
\end{theorem}

\begin{proof}
The definition supplies the object, the notation supplies the property symbol,
and the assumption supplies the condition under which the statement can be read.
\end{proof}

\subsection{Subsection-Level Numbering}\label{sub:SubsectionLevelNumbering}
This subsection shows the second numbering form. Environment counters restart
inside the subsection, and the displayed number includes the subsection index.

\begin{lemma}[sample supporting step]\label{lem:SampleSupportingStep}
The placeholder property is stable under replacing one sample object by another
object from the same finite list.
\end{lemma}

\begin{proposition}[sample transfer]\label{prop:SampleTransfer}
The stability in \cref{lem:SampleSupportingStep} transfers the placeholder
property from \(X_1\) to every \(X_i\) in the list.
\end{proposition}

\begin{corollary}[sample consequence]\label{cor:SampleConsequence}
Every object in the list has the placeholder property.
\end{corollary}

\begin{principle}[sample reading rule]\label{prin:SampleReadingRule}
Template examples should expose the printed convention without importing the
source manuscript's subject matter.
\end{principle}

\begin{axiom}[sample starting premise]\label{ax:SampleStartingPremise}
The sample list is nonempty.
\end{axiom}

\begin{conjecture}[sample future claim]\label{con:SampleFutureClaim}
The same display convention remains legible when the section contains longer
mathematical prose.
\end{conjecture}

\begin{example}[sample use]\label{ex:SampleUse}
The support step is \cref{lem:SampleSupportingStep}. The transfer step is
\cref{prop:SampleTransfer}. The consequence is \cref{cor:SampleConsequence}.
\end{example}

\begin{remark}[section reference form]
The local section helper prints \Sref{sub:SubsectionLevelNumbering}; the formal
cross-reference rule prints the same destination as
\cref{sub:SubsectionLevelNumbering}.
\end{remark}
