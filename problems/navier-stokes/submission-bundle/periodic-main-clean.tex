\section{Conditional Periodic Continuation From the Terminal Endpoint Package}

\begin{theorem}[Conditional periodic continuation]
Let $\nu>0$ and let
\[
  u_0\in C^\infty(\mathbb T^3;\mathbb R^3),
  \qquad \nabla\cdot u_0=0,
  \qquad \int_{\mathbb T^3}u_0\,dx=0.
\]
Then the unforced incompressible Navier--Stokes system
\[
  \partial_tu+u\cdot\nabla u+\nabla p=\nu\Delta u,
  \qquad \nabla\cdot u=0,
  \qquad u(0)=u_0,
\]
on $\mathbb T^3$, and let $(u,p)$ be its maximal classical solution on
$[0,T_*)$.  Assume that the following three inputs hold on this same periodic
solution branch whenever $T_*<\infty$:
\[
\begin{array}{ll}
\mathrm{TTU}:&
\text{terminal-tail uniformization gives a finite retained terminal readout
cover, pointwise field coherence, and a finite-depth transported-center
tower;}\\[0.4em]
\mathrm{END}:&
\text{the endpoint matrix is exhaustive and its four rows}
\\
& Dead,\ packing\text{-}detached,\ tower\text{-}blown,\ Jump
\text{ are removed by participation persistence, pack control,}\\
&\text{terminal tower readout, and field coherence, with cross entries
reduced to those rows;}\\[0.4em]
\mathrm{PATCH}:&
\text{for some }s>5/2\text{ and }N>s+2,\text{ the finite terminal cover and
depth-}N\text{ tower give }
\sup_{t<T_*}\|u(t)\|_{H^s(\mathbb T^3)}<\infty.
\end{array}
\]
Then $T_*=\infty$, and the periodic solution is smooth for all positive time.
\end{theorem}

\begin{proof}
Suppose $T_*<\infty$.  Input \(\mathrm{PATCH}\) gives
\[
  M_s:=\sup_{t<T_*}\|u(t)\|_{H^s(\mathbb T^3)}<\infty
  \qquad (s>5/2).
\]
Classical periodic Navier--Stokes local theory at this fixed \(H^s\) level gives
a continuation time \(\tau=\tau(\nu,s,M_s)>0\) from any time slice with
\(\|u(t)\|_{H^s}\le M_s\).  Choose \(t_j\uparrow T_*\) with
\(T_*-t_j<\tau/2\).  Restarting the same classical branch at \(t_j\) extends it
past \(T_*\), contradicting maximality.  Thus \(T_*=\infty\).  Smoothness on
all of \([0,\infty)\) follows from the smooth local theory and bootstrapping.
\end{proof}
