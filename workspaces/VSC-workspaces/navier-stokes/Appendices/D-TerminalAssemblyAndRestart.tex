\subsection{From terminal rows to one restart datum}

\begin{proposition}[Terminal realization of the rectangle estimate]\label[proposition]{thm:WholeTerminalCompatibleRectangleEstimate}
Let \(0<T_*<\infty\), and let \((u,p)\) be the maximal normalized history on
\(I_*=(0,T_*)\) generated by \((u_0,\nu)\).  Assume
\Cref{thm:BodyWholeTerminalRectangleTheorem}.  For every finite
\(N,M\in\Nzero\),
\begin{equation}\label{eq:WholeTerminalRectangleBound}
\begin{aligned}
  &\sum_{n=0}^{N}
  \norm{V_n}_{L^2(I_*;H^{M+1})}^{2}
  +
  \sum_{n=0}^{N}
  \norm{V_{n+1}}_{L^2(I_*;H^{M-1})}^{2}
  <\infty,
  \\
  &Q_n\in L^1(I_*;H^M),
  \qquad 0\leq n\leq N.
\end{aligned}
\end{equation}
All shared entries are the same distributional derivatives of \(u\), and the
pressure rows retain the normalization \eqref{eq:PressureNormalization}.
\end{proposition}

\begin{proof}
Fix \((N,M)\).  For each nonnegative integrand in
\eqref{eq:BodyRectangleNorm}, monotone convergence and the cutoff-uniform
bound give
\[
  \int_{I_*}f(t)\,dt
  =\lim_{T\uparrow T_*}\int_0^T f(t)\,dt
  <\infty.
\]
The finite sum yields both velocity rows in
\eqref{eq:WholeTerminalRectangleBound}.  The pressure recurrence
\eqref{eq:BodyTemporalPressureRecurrence}, boundedness of the Riesz transforms,
Sobolev multiplication, and Cauchy--Schwarz in time give
\[
  \norm{Q_n}_{L^1(I_*;H^M)}
  \leq
  C_{M,n}\sum_{a=0}^{n}
  \norm{V_a}_{L^2(I_*;H^{M+1})}
  \norm{V_{n-a}}_{L^2(I_*;H^{M+1})}
  <\infty.
\]
Changing the rectangle drops rows or reads them in a lower Sobolev space; it
does not change a shared derivative or normalized pressure.
\end{proof}

\begin{lemma}[The complete spatial row reaches every endpoint topology]\label[lemma]{lem:CompleteSpatialRowEndpoint}
Let \(0<T_*<\infty\), and let \((u,p)\) be the fixed normalized history on
\(I_*=(0,T_*)\).  If
\[
  u\in\bigcap_{m=0}^{\infty}L^2(I_*;H^m(\R^3)),
\]
then for every \(m\in\Nzero\),
\begin{equation}\label{eq:SectionThreeWOneOneEndpoint}
  u_t\in L^1(I_*;H^m),
  \qquad
  u\in W^{1,1}(I_*;H^m)
  \hookrightarrow C([0,T_*];H^m).
\end{equation}
All of these traces meet at one
\[
  u_*:=u(T_*)\in\Hinfty_\sigma(\R^3).
\]
\end{lemma}

\begin{proof}
The pressure-complete transport response is
\[
  (u\cdot\nabla)u+\nabla p
  =\Leray\nabla\cdot(u\otimes u).
\]
Sobolev multiplication and boundedness of the Leray and Riesz multipliers give
\[
  \norm{\Leray\nabla\cdot(u\otimes u)}_{H^m}
  \leq C_m\norm{u}_{H^{m+2}}^2.
\]
Because \(T_*<\infty\), the complete row yields
\begin{align*}
  \norm{\Leray\nabla\cdot(u\otimes u)}_{L^1(I_*;H^m)}
  &\leq C_m\norm{u}_{L^2(I_*;H^{m+2})}^{2}<\infty,\\
  \norm{\Delta u}_{L^1(I_*;H^m)}
  &\leq T_*^{1/2}\norm{u}_{L^2(I_*;H^{m+2})}<\infty.
\end{align*}
The momentum equation puts \(u_t\) in \(L^1(I_*;H^m)\).  For
\(0\leq s<t<T_*\),
\[
  \norm{u(t)-u(s)}_{H^m}
  \leq
  \nu(t-s)^{1/2}\norm{u}_{L^2((s,t);H^{m+2})}
  +C_m\norm{u}_{L^2((s,t);H^{m+2})}^{2}.
\]
Both terms vanish as \(s,t\uparrow T_*\), so the same \(u(t)\) is Cauchy in
every \(H^m\).  The Bochner fundamental theorem of calculus gives the
continuous representatives.  If \(k>m\), the \(H^k\) trace embedded into
\(H^m\) and the native \(H^m\) trace agree almost everywhere and hence
everywhere by continuity.  Their terminal values are one element of
\(\bigcap_mH^m\); divergence and finite energy pass to the limit.
\end{proof}

Assume \Cref{thm:BodyWholeTerminalRectangleTheorem}.  Applying the adjacent
trace to every temporal coordinate gives
\begin{equation}\label{eq:SectionThreeVelocityJetTraces}
  V_N^*:=\lim_{t\uparrow T_*}\partial_t^Nu(t)
  \in\Hinfty(\R^3),
  \qquad N\in\Nzero.
\end{equation}

\begin{proposition}[Pressure-complete endpoint jet]\label[proposition]{prop:PressureCompleteEndpointJet}
Assume the hypotheses of
\Cref{thm:WholeTerminalCompatibleRectangleEstimate}, and let \(u_*\) and
\((V_N^*)_{N\geq0}\) be the common traces furnished by
\Cref{lem:CompleteSpatialRowEndpoint} and
\eqref{eq:SectionThreeVelocityJetTraces}.  Then \(u_*\) determines one sequence
\((V_N^*,Q_N^*)_{N\geq0}\subset\Hinfty\times\Hinfty\) by
\begin{align}
  V_0^*&=u_*,\notag\\
  Q_N^*&=R_iR_j\sum_{a=0}^{N}\binom{N}{a}V_{a,i}^*V_{N-a,j}^*,
  \label{eq:SectionThreeEndpointPressureJet}\\
  V_{N+1}^*&=\nu\Delta V_N^*
  -\sum_{a=0}^{N}\binom{N}{a}(V_a^*\cdot\nabla)V_{N-a}^*
  -\nabla Q_N^*.
  \label{eq:SectionThreeEndpointVelocityJet}
\end{align}
This sequence is the family of left traces of the original normalized
velocity and pressure history.
\end{proposition}

\begin{proof}
The space \(\Hinfty=\bigcap_mH^m\) is stable under multiplication and spatial
differentiation, and Riesz transforms preserve each Sobolev coordinate.  The
two recurrences determine a unique next pair at every finite order.  Fix a
finite target height and take the trace convergence high enough for each
product and derivative.  Passing to the limit in
\eqref{eq:BodyTemporalPressureRecurrence} and
\eqref{eq:BodyTemporalVelocityRecurrence}, then inducting in \(N\), identifies
the generated sequence with the left traces.  The decaying Riesz formula also
preserves the pressure normalization at the terminal face.
\end{proof}

\begin{lemma}[Smooth local restart and uniqueness]\label[lemma]{lem:SmoothLocalRestart}
Let \(t_0\geq0\) and \(a\in\Hinfty_\sigma(\R^3)\).  There are
\(\delta>0\) and a unique smooth normalized pair \((w,q)\) on
\([t_0,t_0+\delta]\times\R^3\), with \(w(t_0)=a\) and
\[
  q=R_iR_j(w_iw_j).
\]
All finite spatial Sobolev orders persist on this same interval, and the
temporal jet at \(t_0\) is generated by
\eqref{eq:SectionThreeEndpointPressureJet}--
\eqref{eq:SectionThreeEndpointVelocityJet}.
\end{lemma}

\begin{proof}
Translate time and apply \Cref{prop:LocalMaximalHistory} to \(a\).  Its
lifespan depends only on \(\nu\) and \(\norm{a}_{H^3}\), while persistence
carries every finite height of \(a\) on that interval.  Uniqueness fixes the
normalized pair.  Differentiating at the initial face and using the Riesz
pressure formula gives the endpoint recurrences.
\end{proof}
