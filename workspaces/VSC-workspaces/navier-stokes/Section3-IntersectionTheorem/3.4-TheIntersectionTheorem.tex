\subsection{The estimate that must survive the endpoint}\label{sub:BodyCompatibleIntersection}

\begin{theorem}[Datum-generated whole-terminal rectangles]\label{thm:BodyWholeTerminalRectangleTheorem}
Let \(u_0\in\Ssigma\), let \(\nu>0\), and let \((u,p)\) be the maximal classical
solution generated by \((u_0,\nu)\) on \([0,T_*)\).  If \(T_*<\infty\), then for every
finite \(N,M\in\Nzero\),
\begin{equation}\label{eq:BodyWholeTerminalRectangleEstimate}
  \sup_{0<T<T_*}\mathfrak R_{N,M}(u;T)<\infty.
\end{equation}
The associated pressure rows are generated by
\eqref{eq:BodyTemporalPressureRecurrence} and use the fixed decaying normalization.
\end{theorem}

\begin{remark}[Reading the terminal quantifier]
The pair \((N,M)\) is fixed before the supremum over \(T<T_*\) is taken.
The bound may depend on \((N,M)\); no constant uniform over all derivative orders is asserted.
Every rectangle belongs to the same history, and larger rectangles return the identical derivatives on their overlaps.
\end{remark}

Everything before this theorem identifies the rectangle on each strict
subinterval.  That is not yet the estimate in
\eqref{eq:BodyWholeTerminalRectangleEstimate}.  Classical smoothness allows the
norm to grow each time the cutoff moves closer to \(T_*\).  The producer proof
has to begin with \((u_0,\nu)\), keep \((N,M)\) fixed, and derive one bound that
does not move with the cutoff.

That derivation is the load-bearing payment of the paper.  The proof source
currently attached to this manuscript states the datum-generated theorem but
does not display the calculation that produces even its first rectangle.  The
terminal argument in \Cref{thm:WholeTerminalCompatibleRectangleEstimate}
starts from \eqref{eq:BodyWholeTerminalRectangleEstimate}; it realizes the
assumed rows on \(I_*\) and cannot serve as their producer.  Until the missing
datum-to-rectangle calculation is inserted here, every conclusion below that
uses \Cref{thm:BodyWholeTerminalRectangleTheorem} remains conditional on this
one theorem.

\Needspace{7\baselineskip}
Fix \(N\) and \(M\).
Every term in \eqref{eq:BodyRectangleNorm} is an integral of a nonnegative squared norm.
For any one integrand \(f\), the intervals \((0,T)\) exhaust \(I_*\), so
\[
  \int_{I_*}f(t)\,dt
  =
  \lim_{T\uparrow T_*}\int_0^T f(t)\,dt.
\]
The whole-terminal estimate keeps that limit finite for every row in the chosen rectangle:
\begin{equation}\label{eq:BodyWholeTerminalAdjacentRows}
  V_n\in L^2(I_*;H^{M+1}),
  \qquad
  V_{n+1}=\partial_tV_n\in L^2(I_*;H^{M-1}),
  \qquad 0\leq n\leq N.
\end{equation}
The pressure part of
\Cref{thm:WholeTerminalCompatibleRectangleEstimate}, using the Riesz bounds and
finite-order Sobolev products, gives the matching rows
\begin{equation}\label{eq:BodyWholeTerminalPressureRows}
  Q_n\in L^1(I_*;H^M),
  \qquad 0\leq n\leq N.
\end{equation}
Together these velocity and pressure coordinates form \(\cR_{N,M}[u_0]\) on the whole interval \(I_*\).

A larger rectangle adds coordinates without replacing the ones already present.
If \(N'\leq N\) and \(M'\leq M\), let \(\rho_{N',M'}^{N,M}\) forget the extra rows and lower the Sobolev height.  Since every retained entry is the same derivative of \(u\),
\begin{equation}\label{eq:BodyRectangleRestrictionCompatibility}
  \rho_{N',M'}^{N,M}\cR_{N,M}[u_0]
  =\cR_{N',M'}[u_0].
\end{equation}
This literal overlap lets the finite blocks be read as one compatible family:
\begin{equation}\label{eq:DatumSpecificCompatibleIntersection}
  \cI[u_0]
  :=\bigl(\cR_{N,M}[u_0]\bigr)_{N,M\in\Nzero},
  \qquad
  \rho_{N',M'}^{N,M}\cR_{N,M}[u_0]
  =\cR_{N',M'}[u_0].
\end{equation}
The intersection creates no new field.
Given any finite temporal order and spatial height, choose one rectangle large enough to contain that coordinate; every larger rectangle returns the same coordinate of the same \(u\).

\begin{proposition}[The mixed intersection]\label[proposition]{prop:BodyCompatibleProjectiveAssembly}
The velocity component of \(\cI[u_0]\) satisfies
\begin{equation}\label{eq:BodyProducedMixedIntersection}
  u\in
  \bigcap_{r=0}^{\infty}
  \bigcap_{m=0}^{\infty}
  H^r(I_*;H^m(\R^3)).
\end{equation}
In particular,
\begin{equation}\label{eq:CompleteSpatialRow}
  u\in\bigcap_{m=0}^{\infty}L^2(I_*;H^m(\R^3)).
\end{equation}
\end{proposition}

\begin{proof}
Fix finite \(r,m\), and choose \(N\geq r\) and \(M\geq m\).
The upper edge of this one rectangle gives
\[
  \partial_t^nu\in L^2(I_*;H^{M+1})
  \hookrightarrow L^2(I_*;H^m)
  \qquad(0\leq n\leq r).
\]
These are the distributional time derivatives of the same velocity field, hence \(u\in H^r(I_*;H^m)\).
Because \(r\) and \(m\) were arbitrary, every finite coordinate belongs to the intersection; \(r=0\) gives \eqref{eq:CompleteSpatialRow}.
\end{proof}
